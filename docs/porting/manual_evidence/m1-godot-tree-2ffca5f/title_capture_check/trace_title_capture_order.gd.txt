extends SceneTree
# Temporary: title_capture.gd's sequence with the portrait state printed at each capture.
const WIDTHS := [88, 99, 124]
func _initialize() -> void:
	var vp := SubViewport.new()
	vp.size = Vector2i(8 + (126 + 8) * WIDTHS.size(), 8 + 172 + 8)
	vp.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(vp)
	var portraits: Array[TitlePortrait] = []
	for k in WIDTHS.size():
		var p := TitlePortrait.new()
		vp.add_child(p)
		p.fit_width(float(WIDTHS[k]))
		portraits.append(p)
	print("after add: frame=%d playing=%s" % [portraits[0].frame, portraits[0].is_playing()])
	var n := portraits[0].sprite_frames.get_frame_count(&"portrait")
	for idx: int in [0, n / 2, n - 1]:
		for p in portraits:
			p.stop()
			p.frame = idx
		print("set %d: frame=%d playing=%s" % [idx, portraits[0].frame, portraits[0].is_playing()])
		for i in 4:
			await process_frame
			print("  after process_frame %d: frame=%d playing=%s" % [i, portraits[0].frame, portraits[0].is_playing()])
		await RenderingServer.frame_post_draw
		print("  at capture: frame=%d playing=%s" % [portraits[0].frame, portraits[0].is_playing()])
	quit(0)
