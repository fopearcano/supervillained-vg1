extends SceneTree
# Temporary: every title frame in title_capture.gd's layout, set after the portraits are ready.
const WIDTHS := [88, 99, 124]
var out := ""
func _initialize() -> void:
	out = OS.get_cmdline_user_args()[0]
	_run.call_deferred()
func _run() -> void:
	DirAccess.make_dir_recursive_absolute(out)
	var vp := SubViewport.new()
	vp.size = Vector2i(8 + (126 + 8) * WIDTHS.size(), 8 + 172 + 8)
	vp.transparent_bg = false
	vp.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(vp)
	var bg := ColorRect.new()
	bg.color = Color("#120818")
	bg.size = Vector2(vp.size)
	vp.add_child(bg)
	var portraits: Array[TitlePortrait] = []
	for k in WIDTHS.size():
		var p := TitlePortrait.new()
		vp.add_child(p)
		p.fit_width(float(WIDTHS[k]))
		p.position = Vector2(8 + k * (126 + 8), 8)
		portraits.append(p)
	await process_frame
	var n: int = portraits[0].sprite_frames.get_frame_count(&"portrait")
	for idx in n:
		for p in portraits:
			p.stop()
			p.frame = idx
		for i in 4:
			await process_frame
		await RenderingServer.frame_post_draw
		var img := vp.get_texture().get_image()
		img.resize(img.get_width() * 2, img.get_height() * 2, Image.INTERPOLATE_NEAREST)
		img.save_png(out.path_join("f%02d.png" % idx))
	quit(0)
